# Everything on this site is meant to be read by people, search engines and
# answer engines alike. The wildcard rule already permits every crawler; the
# named entries below make the AI-crawler policy explicit so a future edit
# cannot restrict one of them by accident, and so a crawler that only honours
# its own token still finds an explicit Allow.

User-agent: *
Allow: /

Allow: /

Allow: /

Allow: /

# Google-Extended controls whether Google may use pages for Gemini training
# and grounding. It does NOT affect Search ranking or AI Overviews, which use
# Googlebot. Decision: ALLOW. Being in the grounding corpus is how a small
# domain becomes a resolvable entity for Gemini; blocking it would give up
# that with nothing gained in Search.
User-agent: Google-Extended
Allow: /

# Apple: Applebot-Extended is the opt-out token for Apple Intelligence
# training. Allowed for the same reason as Google-Extended.
User-agent: Applebot-Extended
Allow: /

# Common Crawl feeds most open training sets. Allowed.
User-agent: CCBot
Allow: /

# Content Signals (contentsignals.org, draft-romm-aipref-contentsignals).
# Deliberate policy, not a default: this site wants to be in training corpora
# and in AI answers. Revisit only if that stops being the strategy.
Content-Signal: ai-train=yes, search=yes, ai-input=yes
